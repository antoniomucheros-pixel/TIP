#include <iostream>

int main() {
    int a, b;
    std::cin >> a >> b;
    a = a - b;
    b = a + b;
    a = b - a;
    std::cout << a << "\n" << b << "\n";
    return 0;
}

#include <iostream>

int main() {
    int n;
    int t = 1;
    int s = 1;
    std::cin >> n;
    for (int i = 1; i <= n; i++) {
        t = t * i;
        s = s * t;
    }
    std::cout << s << "\n";
    return 0;
}
